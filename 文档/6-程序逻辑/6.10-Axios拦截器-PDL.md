# 6.10 Axios拦截器算法 — PDL语言

```pdl
// ============ 请求拦截器 ============
PROCEDURE requestInterceptor(config)
    token ← localStorage.getItem("token")
    IF token EXISTS THEN
        config.headers["Authorization"] ← "Bearer " + token
    END IF
    RETURN config
END PROCEDURE

// ============ 响应拦截器 ============
PROCEDURE responseInterceptor(response)
    // 正常响应
    RETURN response.data
END PROCEDURE

PROCEDURE responseErrorInterceptor(error)
    status ← error.response.status

    CASE status OF
        401:  // 未授权/Token过期
            localStorage.removeItem("token")
            localStorage.removeItem("userInfo")
            router.push("/login")
            SHOW message("登录已过期，请重新登录")

        403:  // 无权限
            SHOW message("无访问权限")

        500:  // 服务器错误
            SHOW message("服务器错误，请稍后重试")

        DEFAULT:
            SHOW message(error.response.data.message OR "请求失败")
    END CASE

    RETURN Promise.reject(error)
END PROCEDURE
```

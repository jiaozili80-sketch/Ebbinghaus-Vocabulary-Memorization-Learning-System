# 6.4 学习反馈处理算法 — PDL语言（过程设计语言）

```pdl
PROCEDURE study_submit(wordId, familiarity, duration)
    // 获取当前登录用户
    userId ← LoginUserContext.getUserId()

    // 查询复习计划
    reviewPlan ← ReviewPlanMapper.selectByUserAndWord(userId, wordId)

    IF reviewPlan IS NULL THEN
        RETURN error("该单词未在学习计划中")
    END IF

    // 获取当前阶段
    currentStage ← reviewPlan.getStage()

    // ============ 核心算法：艾宾浩斯间隔重复 ============
    // 根据熟悉度计算新阶段
    CASE familiarity OF
        1:  // 不认识
            newStage ← 0
        2:  // 模糊
            newStage ← MAX(0, currentStage - 1)
        3:  // 认识
            newStage ← MIN(MAX_INTERVAL_INDEX, currentStage + 1)
    END CASE

    // 读取间隔时间配置
    intervals ← config.getReviewIntervals()
    // intervals = [5, 30, 720, 1440, 2880, 5760, 10080, 21600]

    // 计算下次复习时间
    nextReviewTime ← NOW() + intervals[newStage] * MINUTES

    // 判断是否已掌握
    IF familiarity >= 3 AND newStage = MAX_INTERVAL_INDEX THEN
        status ← 1    // 已掌握
    ELSE
        status ← 0    // 继续复习
    END IF

    // 更新复习计划
    reviewPlan.setStage(newStage)
    reviewPlan.setNextReviewTime(nextReviewTime)
    reviewPlan.setStatus(status)
    reviewPlan.setFamiliarity(familiarity)
    ReviewPlanMapper.updateById(reviewPlan)

    // ============ 错词处理 ============
    IF familiarity <= 2 THEN
        wrongWord ← WrongWordMapper.selectByUserAndWord(userId, wordId)
        IF wrongWord EXISTS THEN
            wrongWord.setWrongCount(wrongWord.getWrongCount() + 1)
            WrongWordMapper.updateById(wrongWord)
        ELSE
            newWrongWord ← new WrongWord(userId, wordId, 1)
            WrongWordMapper.insert(newWrongWord)
        END IF
    END IF

    // ============ 学习记录更新 ============
    today ← DATE(NOW())
    studyRecord ← StudyRecordMapper.selectByUserAndDate(userId, today)

    IF studyRecord IS NULL THEN
        studyRecord ← new StudyRecord()
        studyRecord.setUserId(userId)
        studyRecord.setStudyDate(today)
        studyRecord.setNewWordCount(0)
        studyRecord.setReviewWordCount(0)
        studyRecord.setStudyDuration(0)
    END IF

    IF isNewWord(reviewPlan) THEN
        studyRecord.setNewWordCount(studyRecord.getNewWordCount() + 1)
    ELSE
        studyRecord.setReviewWordCount(studyRecord.getReviewWordCount() + 1)
    END IF

    studyRecord.setStudyDuration(studyRecord.getStudyDuration() + duration)
    StudyRecordMapper.saveOrUpdate(studyRecord)

    // ============ 用户统计更新 ============
    user ← UserMapper.selectById(userId)
    user.setTotalWords(user.getTotalWords() + 1)
    IF status = 1 THEN
        user.setMasteredWords(user.getMasteredWords() + 1)
    END IF
    UserMapper.updateById(user)

    // ============ 返回结果 ============
    result ← new StudySubmitVO()
    result.setNewStage(newStage)
    result.setNextReviewTime(nextReviewTime)
    result.setIsMastered(status = 1)

    RETURN success(result)

END PROCEDURE
```
